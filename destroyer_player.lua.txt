-- ====== Maid ======
local Maid = {}
Maid.__index = Maid
function Maid.new() return setmetatable({_tasks = {}, _destroyed = false}, Maid) end
function Maid:GiveTask(task)
    if self._destroyed then self:_cleanupTask(task) return end
    table.insert(self._tasks, task)
    return task
end
function Maid:_cleanupTask(task)
    local t = typeof(task)
    if t == "RBXScriptConnection" then task:Disconnect()
    elseif t == "Instance" then task:Destroy()
    elseif t == "function" then task()
    elseif t == "thread" then pcall(function() task.cancel(task) end)
    elseif t == "table" and type(task.Destroy) == "function" then task:Destroy()
    end
end
function Maid:DoCleaning()
    if self._destroyed then return end
    self._destroyed = true
    for _, task in ipairs(self._tasks) do self:_cleanupTask(task) end
    self._tasks = {}
end
function Maid:Destroy() self:DoCleaning() end

local function guiRoots()
    local roots, seen = {}, {}
    local function add(r)
        if typeof(r) == "Instance" and not seen[r] then seen[r] = true; roots[#roots + 1] = r end
    end
    pcall(function() add(gethui and gethui()) end)
    pcall(function() add(getcore and getcore()) end)
    add(CoreGui)
    pcall(function() add(LocalPlayer:FindFirstChildOfClass("PlayerGui")) end)
    return roots
end

local function cardHasStop(node, stopSet)
    for _, d in ipairs(node:GetDescendants()) do
        if d:IsA("TextLabel") then
            if stopSet[d.Text] then return true end
            for _, w in ipairs(BAD_WORDS) do
                if d.Text:find(w, 1, true) then return true end
            end
        end
    end
    return false
end

local function cardOf(label, stopSet)
    local node = label.Parent
    for _ = 1, 7 do
        if not node or typeof(node) ~= "Instance" or node == game then return nil end
        if node:IsA("Frame") or node:IsA("ScrollingFrame") then
            local framed = node:FindFirstChildOfClass("UIStroke") or node:FindFirstChildOfClass("UICorner")
            if framed then
                local sz = node.AbsoluteSize
                if sz.Y > 0 and sz.Y < 520 and not cardHasStop(node, stopSet) then
                    return node
                end
            end
        end
        node = node.Parent
    end
    return nil
end

local function scanAndKill(titleSet, stopSet)
    local killed, visited = {}, {}
    for _, root in ipairs(guiRoots()) do
        local labels = {}
        for _, d in ipairs(root:GetDescendants()) do
            if d:IsA("TextLabel") then labels[#labels + 1] = d end
        end
        for _, label in ipairs(labels) do
            if label.Parent and titleSet[label.Text] then
                local card = cardOf(label, stopSet)
                if card and not killed[card] and not visited[card] then
                    visited[card] = true
                    killed[card] = true
                    pcall(function() card:Destroy() end)
                end
            end
        end
    end
    local n = 0
    for _ in pairs(killed) do n = n + 1 end
    return n
end

local function bind_safecallback(callback)
    if not callback then return end
    local ok, err = xpcall(callback, function(e) return debug.traceback(e) end)
    if not ok then warn("[BIND ERROR] " .. tostring(err)) end
end

local function Bind_GetStorage()
    local parent
    local ok, res = pcall(function()
        if gethui then return gethui() end
        if getcore then return getcore() end
    end)
    if ok and typeof(res) == "Instance" then parent = res else parent = CoreGui end
    if typeof(parent) ~= "Instance" then parent = LocalPlayer:FindFirstChildOfClass("PlayerGui") end
    if typeof(parent) ~= "Instance" then parent = LocalPlayer:WaitForChild("PlayerGui", 5) end
    if typeof(parent) ~= "Instance" then parent = CoreGui end

        sg = Instance.new("ScreenGui")
        sg.Name = "@bindstorage_v5"

        sg.IgnoreGuiInset = true
        pcall(function() sg.ScreenInsets = Enum.ScreenInsets.None end)
        if syn and syn.protect_gui then pcall(syn.protect_gui, sg) end
        sg.Parent = parent

end

local function Bind_MakeDraggable(gui, maid, ripple, sound, clickFunc)
    local dragging, dragInput, dragStart, startPos, hasMoved = false, nil, nil, nil, false

    maid:GiveTask(gui.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging, dragStart, startPos, hasMoved = true, input.Position, gui.Position, false
            pcall(function() sound:Play() end)

            local absPos = gui.AbsolutePosition
            ripple.Position = __UD2(0, input.Position.X - absPos.X, 0, input.Position.Y - absPos.Y)
            ripple.Size = __UD2(0, 0, 0, 0)
            ripple.BackgroundTransparency = 0.45
            ripple.Visible = true
            TweenService:Create(ripple, TweenInfo.new(0.4, Enum.EasingStyle.Sine, Enum.EasingDirection.Out), {
                Size = __UD2(0, 45, 0, 45), BackgroundTransparency = 1
            }):Play()

            local rel
            rel = UserInputService.InputEnded:Connect(function(endInput)
                if endInput.UserInputType == input.UserInputType then
                    dragging = false
                    if not hasMoved then bind_safecallback(clickFunc) end
                    rel:Disconnect()
                end
            end)
        end
    end))

    maid:GiveTask(gui.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end))

    maid:GiveTask(UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            if delta.Magnitude > 7 then hasMoved = true end
            local prt = gui.Parent
            if prt then
                local screen = prt.AbsoluteSize
                gui.Position = __UD2(startPos.X.Scale + (delta.X / screen.X), 0, startPos.Y.Scale + (delta.Y / screen.Y), 0)
            end
        end
    end))
end

local roleCache = { data = nil, timestamp = 0, TTL = 0.8 }
local function getCachedRoleData()
    local now = os.clock()
    if roleCache.data and (now - roleCache.timestamp) < roleCache.TTL then return roleCache.data end
    local ok, result = pcall(function()
        local remote = ReplicatedStorage:FindFirstChild("GetPlayerData", true)
        if remote and remote:IsA("RemoteFunction") then return remote:InvokeServer() end
    end)
    if ok and result then
        roleCache.data, roleCache.timestamp = result, now
        return result
    end
    roleCache.timestamp = now
    return roleCache.data
end

local function isValidTarget(player)
    return player and player ~= LocalPlayer and player.Parent
        and not state.whitelist[player.UserId]
        and player.Character and player.Character:FindFirstChild("HumanoidRootPart")
end

local function getSelectedOrFirst()
    if isValidTarget(state.resetSelPlr) then return state.resetSelPlr end
    for _, player in ipairs(state.selectedPlayers) do
        if isValidTarget(player) then return player end
    end
    return nil
end

local function quickScanSheriff()
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            local char = player.Character
            if char then
                if char:FindFirstChild("Gun") or char:FindFirstChild("Revolver") then
                    if not state.whitelist[player.UserId] then return player end
                end
                local tool = char:FindFirstChildOfClass("Tool")
                if tool then
                    local name = tool.Name:lower()
                    if name:find("gun") or name:find("revolver") then
                        if not state.whitelist[player.UserId] then return player end
                    end
                end
            end
            local bp = player:FindFirstChild("Backpack")
            if bp and (bp:FindFirstChild("Gun") or bp:FindFirstChild("Revolver")) then
                if not state.whitelist[player.UserId] then return player end
            end
        end
    end
    return nil
end

local function findTargetByRole(roleName)
    local roleData = getCachedRoleData()
    if roleData then
        for playerName, data in pairs(roleData) do
            if data.Role == roleName and not data.Killed and not data.Dead then
                local p = Players:FindFirstChild(playerName)
                if p and p ~= LocalPlayer and not state.whitelist[p.UserId] then return p end
            end
        end
    end
    return nil
end

local function findSheriff()
    local quick = quickScanSheriff()
    if quick and quick ~= LocalPlayer and not state.whitelist[quick.UserId] then return quick end
    return findTargetByRole("Sheriff")
end
local function findMurderer() return findTargetByRole("Murderer") end

        Workspace.FallenPartsDestroyHeight = originalDestroyHeight
        if not character or not savedData then return end
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        local rootPart = character:FindFirstChild("HumanoidRootPart")
        if not humanoid or not rootPart then return end
        rootPart.CFrame = savedData.cframe
        rootPart.AssemblyLinearVelocity = Vector3.zero
        rootPart.AssemblyAngularVelocity = Vector3.zero
        humanoid.PlatformStand = false
        humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
        if humanoid.Health < humanoid.MaxHealth then humanoid.Health = humanoid.MaxHealth end
        for _, part in ipairs(character:GetDescendants()) do
            if part:IsA("BasePart") then part.CanCollide = true end
        end
    end)
    pcall(function() Workspace.FallenPartsDestroyHeight = originalDestroyHeight end)
end

-- ====== Главная функция сброса ======
local function VoidReset(TargetPlayer, _retryCount)
    if not TargetPlayer or not TargetPlayer.Parent then return end
    if TargetPlayer == LocalPlayer or state.whitelist[TargetPlayer.UserId] then return end
    _retryCount = _retryCount or 0
    cancelCurrentReset()

    local Character = LocalPlayer.Character
    if not Character then return end
    local Humanoid = Character:FindFirstChildOfClass("Humanoid")
    local RootPart = Humanoid and Humanoid.RootPart
    local TCharacter = TargetPlayer.Character
    local TRootPart = TCharacter and TCharacter:FindFirstChild("HumanoidRootPart")
    local THead = TCharacter and TCharacter:FindFirstChild("Head")
    if not (Humanoid and RootPart and TRootPart) then return end

    local touchParts = {}
    local partPriority = {"RightHand","LeftHand","RightFoot","LeftFoot","Right Arm","Left Arm","Right Leg","Left Leg","Torso","HumanoidRootPart","Head"}
    for _, name in ipairs(partPriority) do
        local p = TCharacter:FindFirstChild(name)
        if p and p:IsA("BasePart") then table.insert(touchParts, p) end
    end
    if #touchParts == 0 then table.insert(touchParts, TRootPart) end

    local savedData = { cframe = RootPart.CFrame }
    local originalDestroyHeight = Workspace.FallenPartsDestroyHeight
    Workspace.FallenPartsDestroyHeight = -math.huge
    Humanoid.PlatformStand = true

    local bv = Instance.new("BodyVelocity")
    bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
    bv.Velocity = Vector3.new(0, -200000, 0)
    bv.Parent = RootPart

    local bg = Instance.new("BodyGyro")
    bg.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
    bg.P = 9e8; bg.D = 9e8
    bg.CFrame = RootPart.CFrame
    bg.Parent = RootPart

    local startTime = os.clock()
    local frameCount = 0
    local done = false
    local resetObj = { bv = bv, bg = bg, conn = nil }

    local function cleanup(success, manual)
        if done then return end
        done = true
        if currentReset == resetObj then currentReset = nil end
        if resetObj.conn then pcall(function() resetObj.conn:Disconnect() end) end
        pcall(function() bv:Destroy() end)
        pcall(function() bg:Destroy() end)
        restoreSelf(Character, savedData, originalDestroyHeight)
        StatusHUD.Set("idle")
        BindableButtons.ResetActive = false
        if not success and not manual and _retryCount < config.maxRetries then
            task.delay(config.retryDelay, function()
                if TargetPlayer and TargetPlayer.Parent and not state.whitelist[TargetPlayer.UserId] then
                    VoidReset(TargetPlayer, _retryCount + 1)
                end
            end)
        end
    end

    resetObj.cancel = function() cleanup(true, true) end
    currentReset = resetObj

    StatusHUD.Set("active", TargetPlayer.Name)
    BindableButtons.ResetActive = true

    resetObj.conn = RunService.Heartbeat:Connect(function()
        frameCount += 1
        if TargetPlayer.Character ~= TCharacter or not TRootPart.Parent then cleanup(true) return end
        if os.clock() - startTime >= config.resetDuration or not Character.Parent or not RootPart.Parent then cleanup(false) return end

        local angle = (os.clock() - startTime) * (math.pi * 20)
        local dynamicOffset = Vector3.new(math.cos(angle) * 1.5, math.sin(angle) * 1.5, math.sin(angle) * 1.5)
        RootPart.CFrame = CFrame.new(TRootPart.Position + dynamicOffset)
        RootPart.AssemblyLinearVelocity = Vector3.new(200000, -200000, 200000)
        RootPart.AssemblyAngularVelocity = Vector3.new(20000, 20000, 20000)

        if frameCount % 2 == 1 then
            for _ = 1, 2 do
                for _, part in ipairs(touchParts) do touch(RootPart, part) end
            end
        else
            touch(RootPart, TRootPart)
            if THead then touch(RootPart, THead) end
        end

        if type(sethiddenproperty) == "function" then
            pcall(sethiddenproperty, RootPart, "PhysicsRepRootPart", TRootPart)
        end
        if Humanoid.Health < Humanoid.MaxHealth * 0.5 then
            pcall(function() Humanoid.Health = Humanoid.MaxHealth end)
        end
    end)
end

-- ====== Авто-сброс ======
local function setupAutoTarget(maidKey, finderFunc, interval)
    return function(enabled)
        if maids[maidKey] then maids[maidKey]:Destroy() end
        if enabled then
            maids[maidKey] = Maid.new()
            local active = true
            maids[maidKey]:GiveTask(function() active = false end)
            task.spawn(function()
                while active do
                    local target = nil
                    pcall(function()
                        target = finderFunc()
                        if target then VoidReset(target) end
                    end)
                    if target then task.wait(math.max(interval, config.resetDuration + 0.08))
                    else task.wait(interval) end
                end
            end)
        end
    end
end
